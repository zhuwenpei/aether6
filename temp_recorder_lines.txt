   924	  // Helper to fix MP4 container duration header so mobile photo albums and media players display exact duration

   926	    try {
   927	      const buffer = await blob.arrayBuffer();
   928	      const bytes = new Uint8Array(buffer);

   930	
   931	      for (let i = 0; i < bytes.length - 32; i++) {
   932	        // Search for 'mvhd' box (Movie Header Box)

   934	          const version = bytes[i + 4];
   935	          if (version === 0) {
   936	            const timescale = view.getUint32(i + 16, false);
   937	            if (timescale > 0) {
   938	              view.setUint32(i + 20, Math.round(durationSec * timescale), false);
   939	            }
   940	          } else if (version === 1) {
   941	            const timescale = view.getUint32(i + 24, false);

   945	            }
   946	          }
   947	        }
   948	        // Search for 'mdhd' box (Media Header Box)
   949	        if (bytes[i] === 0x6d && bytes[i+1] === 0x64 && bytes[i+2] === 0x68 && bytes[i+3] === 0x64) {
   950	          const version = bytes[i + 4];
   951	          if (version === 0) {
   952	            const timescale = view.getUint32(i + 16, false);

   961	            }

   981	  // Helper to fix WebM duration header so media players display exact duration
   982	  const fixWebmDuration = async (blob: Blob, durationMs: number): Promise<Blob> => {
   983	    try {
   984	      const buffer = await blob.arrayBuffer();
   985	      const bytes = new Uint8Array(buffer);
   986	      
   987	      // Search for Segment Info ID: 0x15, 0x49, 0xA9, 0x66
   988	      for (let i = 0; i < bytes.length - 16; i++) {
   989	        if (bytes[i] === 0x15 && bytes[i+1] === 0x49 && bytes[i+2] === 0xa9 && bytes[i+3] === 0x66) {
   990	          let foundDuration = false;
   991	          for (let j = i + 4; j < i + 200 && j < bytes.length - 8; j++) {
   992	            if (bytes[j] === 0x44 && bytes[j+1] === 0x89) {
   993	              foundDuration = true;
   994	              const length = bytes[j+2];
   995	              if (length === 8) {
   996	                const view = new DataView(buffer, j + 3, 8);
   997	                view.setFloat64(0, durationMs, false);
   998	                return new Blob([buffer], { type: blob.type });
   999	              } else if (length === 4) {
  1000	                const view = new DataView(buffer, j + 3, 4);
  1001	                view.setFloat32(0, durationMs, false);
  1002	                return new Blob([buffer], { type: blob.type });
  1003	              }
  1004	            }
  1005	          }
  1006	
  1007	          // If Duration tag was missing, inject 0x44 0x89 0x88 [Float64 durationMs] into Segment Info
  1008	          if (!foundDuration && i + 8 < bytes.length) {
  1009	            const insertPos = i + 8; // Right inside Segment Info
  1010	            const durTag = new Uint8Array(11);
  1011	            durTag[0] = 0x44; // Element ID Duration
  1012	            durTag[1] = 0x89;
  1013	            durTag[2] = 0x88; // Length 8 bytes
  1014	            const view = new DataView(durTag.buffer, 3, 8);
  1015	            view.setFloat64(0, durationMs, false);
  1016	
  1017	            const newBuffer = new Uint8Array(bytes.length + 11);
  1018	            newBuffer.set(bytes.subarray(0, insertPos), 0);
  1019	            newBuffer.set(durTag, insertPos);
  1020	            newBuffer.set(bytes.subarray(insertPos), insertPos + 11);
  1021	
  1022	            return new Blob([newBuffer.buffer], { type: blob.type });
  1023	          }
  1024	        }
  1025	      }
  1026	    } catch (e) {
  1027	      console.warn("Could not patch EBML duration header:", e);
  1028	    }
